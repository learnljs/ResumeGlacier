# 知识技能评分细化 — 开发方案

> **目标：** 基于《知识技能评分细化》需求文档，将知识和技能维度评分从「描述性星级判定」升级为「31 题逐题回答 → 后端计算」的精确评分系统，提升评分的区分度和可解释性。

**架构：** AI 负责语义理解（解析 JD + 回答 31 个是非题 + 识别项目类型），后端 TypeScript 负责机械计算（映射表 → 多数决 → 保险规则 → 最终星级）。不新增数据库字段，不改变前端 UI。

**技术栈：** TypeScript, Prisma/SQLite, OpenAI API

## 全局约束

- 评分规则严格按需求文档的 5 步流程执行
- AI 只做 Step 1-2，后端做 Step 3-5
- 无 JD 场景与有 JD 场景共用同一套 31 题清单
- 不修改数据库 schema（`aiDetail` 字段已存在）
- 前端不做任何改动
- 旧版评分结果不做数据迁移，靠 `aiDetail` 中有无 `questions` 字段区分新旧格式
- 筛选并发度保持 `SCREENING_CONCURRENCY=2` 不变

### 3.4 多数决规则（需求文档 4.3）

- 4 个方向中出现次数最多的星级 = 维度最终星级

- 降至 1 星：所有 4 个方向都 ≤ 2 星，否则升为 2 星
- 平局取低
- 侧重视描述：某方向简历未提及不因此降级

### Task 1: 重写评分标准常量

**文件：**
- 重写：`lib/prompts/knowledge-skill-criteria.ts`

**说明：** 删除旧的 `FULL_CRITERIA` 和 `GENERIC_CRITERIA`，新增一个函数 `buildDetailedCriteria()` 生成 31 题清单文本，供 prompt 拼装使用。有 JD 和无 JD 共用同一套清单，无 JD 时在 prompt 外层加一句说明即可。

- [ ] **Step 1: 重写文件**

`lib/prompts/knowledge-skill-criteria.ts` 完整内容：

```typescript
/**
 * 知识/技能维度细化评分标准（31 题版）
 *
 * 职责：生成 31 题清单文本，供 prompt 拼装使用。
 * AI 负责逐题回答是/否并引用简历原文。
 * 后端负责映射表 → 多数决 → 保险规则计算最终星级。
 */

/**
 * 生成 31 题详细评分标准（供 AI prompt 使用）
 *
 * @param hasJD 是否有岗位 JD（影响 prompt 中的 JD 相关说明）
 * @returns 31 题清单 markdown 文本
 */
export function buildDetailedCriteria(hasJD: boolean): string {
  const jdNote = hasJD
    ? `**JD 相关说明：**
- 若 JD 明确「学历专业不限」或对学历层次无硬性要求，Q5/Q6 仍按候选人实际专业/学历情况正常回答
- 若 JD 有经验年限要求，Q1 以 JD 要求为准；JD 未提则默认 3 年
- JD 提到某方面：评分时重点检查简历是否符合该 JD 要求
- JD 没提到某方面：正常判断，基于简历内容回答，不视为"否"
- JD 中"XX经验者优先"不视为加分项，有则正常算分，无则正常不算分
`
    : `**通用评估说明：**
- Q1 年限标准统一为 3 年
- 所有问题基于简历信号的绝对强度判断，不参考岗位要求
`

  return `## 评分规则总原则

1. 评分时结合 JD 重点，但所有问题都适用。JD 没提到的内容正常基于简历内容回答，**不视为"否"**
2. 每题答"是"时必须引用简历原文作为 evidence，答"否"时 evidence 为 null
3. 同一段简历描述可被多个问题引用并都答"是"，不视为重复计分
4. 评分结果供 HR 参考，HR 可随时修改星级

${jdNote}

## 重要判定规则

1. **技能清单自评词全部忽略**：简历中"了解/熟悉/熟练/精通/掌握"等自评词一律不作为 Q9-Q11 和 Q20-Q23 的评分依据。只有工具出现在实践描述中有具体使用场景才算"是"。

2. **模糊表述不算深度**："提升效率""优化体验""提高性能"等无具体数字或对比基线的描述，在 Q24/Q25 中视为"否"。
```

```typescript
}

/**
 * 计算知识和技能维度的最终星级
 */
export function calculateScores(input: ScorerInput): ScorerOutput {
  const knowledge = scoreDimension(
    'knowledge',
    input.knowledgeQuestions,
    KNOWLEDGE_DIRECTIONS,
    input.projectType
  )
  const skill = scoreDimension(
    'skill',
    input.skillQuestions,
    SKILL_DIRECTIONS,
    input.projectType
  )
  return { knowledge, skill }
}

// === 维度评分核心 ===

function scoreDimension(
  dimName: 'knowledge' | 'skill',
  questions: Record<string, QuestionAnswer>,
  directions: Record<string, string[]>,
  projectType: ProjectType
): ScoredDimension {
  // 容错：如果整个维度的问题都缺失，返回 0
  const allEmpty = Object.values(directions)
    .flat()
    .every((qId) => !questions[qId])
  if (allEmpty) {
    return {
      rating: 0,
      questions: {},
      directions: {},
      voting: { directionStars: [], majorityStar: 0, adjustments: ['所有问题缺失，无法计算'] },
      insurance: { applied: [], overrides: [] },
    }
  }

  // Step 3: 映射表 → 每个方向的基础星级
  const directionScores: Record<string, DirectionScore> = {}
  for (const [dirKey, qIds] of Object.entries(directions)) {
    const yesCount = qIds.filter((qId) => questions[qId]?.answer === true).length
    const totalCount = qIds.length
    const star = mapYesCountToStar(yesCount, totalCount)
    directionScores[dirKey] = { yesCount, totalCount, star }
  }

  // Step 4: 多数决
  const directionStars = Object.values(directionScores).map((d) => d.star)
  const voting = applyMajorityVote(directionStars, dimName)

  // Step 5: 保险规则
  const insurance = applyInsuranceRules(
    voting.majorityStar,
    directionScores,
    questions,
    directions,
    projectType,
    dimName
  )

  const finalRating = insurance.overrides.length > 0
    ? // 取最后一次 override 的值（按保险规则优先级）
      resolveOverriddenRating(insurance.overrides, voting.majorityStar)
    : voting.majorityStar

  return {
    rating: finalRating,
    questions,
    directions: directionScores,
    voting,
    insurance,
  }
}

// === Step 3: 映射表 ===

/**
 * 基础映射表（需求文档 4.2）：
 * 4 问题：0→1, 1→2, 2→3, 3→4, 4→5
 * 3 问题（仅工具方法认知）：0→1, 1→2, 2→4, 3→5
 */
function mapYesCountToStar(yesCount: number, totalCount: number): number {
  if (totalCount === 3) {
    // 工具方法认知方向
    switch (yesCount) {
      case 0: return 1
      case 1: return 2
      case 2: return 4
      case 3: return 5
      default: return 1
    }
  }
  // 4 问题方向
  return Math.min(yesCount + 1, 5)
}

// === Step 4: 多数决 ===

function applyMajorityVote(
  stars: number[],
  dimName: 'knowledge' | 'skill'
): {
  directionStars: number[]
  majorityStar: number
  adjustments: string[]
} {
  const adjustments: string[] = []

  // 计算每个星级的出现次数
  const counts = new Map<number, number>()
  for (const s of stars) {
    counts.set(s, (counts.get(s) || 0) + 1)
  }

  // 多数决：出现次数最多的星级
  let maxCount = 0
  let majorityStar = 1
  const ties: number[] = []
  for (const [star, count] of counts) {
    if (count > maxCount) {
      maxCount = count
      majorityStar = star
      ties.length = 0
      ties.push(star)
    } else if (count === maxCount) {
      ties.push(star)
    }
  }

  // 平局取低
  if (ties.length > 1) {
    const lower = Math.min(...ties)
    if (lower !== majorityStar) {
      adjustments.push(`平局取低：${ties.join('星和')}星持平，取${lower}星`)
      majorityStar = lower
    }
  }

  // 升至 5 星多数决失效：至少 3 个方向达 5 星
  if (majorityStar === 5) {
    const fiveStarCount = stars.filter((s) => s === 5).length
    if (fiveStarCount < 3) {
      adjustments.push(
        `升至5星多数决失效：仅${fiveStarCount}个方向达5星，不满足≥3个的要求，降为4星`
      )
      majorityStar = 4
    }
  }

  // 降至 1 星多数决失效：必须全部 ≤ 2 星
  if (majorityStar === 1) {
    const allLow = stars.every((s) => s <= 2)
    if (!allLow) {
      adjustments.push('降至1星多数决失效：存在方向>2星，升为2星')
      majorityStar = 2
    }
  }

  return { directionStars: stars, majorityStar, adjustments }
}

// === Step 5: 保险规则 ===

interface InsuranceResult {
  applied: string[]
  overrides: string[]
}

function applyInsuranceRules(
  star: number,
  directionScores: Record<string, DirectionScore>,
  questions: Record<string, QuestionAnswer>,
  directions: Record<string, string[]>,
  projectType: ProjectType,
  dimName: 'knowledge' | 'skill'
): InsuranceResult {
  const applied: string[] = []
  const overrides: string[] = []
  let currentStar = star

  // 1星硬门槛（必须全部方向都是0个"是"）
  if (currentStar === 1) {
    const allZero = Object.values(directionScores).every((d) => d.yesCount === 0)
    if (!allZero) {
      applied.push('1星硬门槛触发：存在"是"的答案，不符合1星条件')
      currentStar = 2
      overrides.push('1星→2星（1星硬门槛不满足）')
    }
  }

  // 5星硬门槛（获得5星的方向必须全"是"）
  if (currentStar === 5) {
    const fiveStarDirections = Object.entries(directionScores)
      .filter(([, d]) => d.star === 5)
    const allFiveStarFull = fiveStarDirections.length > 0 &&
      fiveStarDirections.every(([, d]) => d.yesCount >= d.totalCount)
    if (!allFiveStarFull) {
      applied.push('5星硬门槛触发：存在5星方向未满"是"，不符合5星条件')
      currentStar = 4
      overrides.push('5星→4星（5星硬门槛不满足）')
    }
  }

  // 项目类型规则（仅技能维度）
  if (dimName === 'skill') {
    const meetsQ18 = questions['Q18']?.answer === true
    const enterpriseCount = projectType.enterprise || 0
    const nationalCount = projectType.nationalCompetition || 0

    // 4星：需 Q18 命中
    if (currentStar >= 4 && !meetsQ18) {
      applied.push('项目类型规则触发：4星需Q18命中（企业级/长期实习项目），不满足')
      currentStar = 3
      overrides.push(`${'4星'}→3星（项目类型不满足Q18）`)
    }

    // 5星：需 2 段企业级 或 1 段企业级 + 1 段国家级/国际级竞赛
    if (currentStar === 5) {
      const hasTwoEnterprise = enterpriseCount >= 2
      const hasEnterprisePlusNational = enterpriseCount >= 1 && nationalCount >= 1
      if (!hasTwoEnterprise && !hasEnterprisePlusNational) {
        applied.push('项目类型规则触发：5星需2段企业级或1段企业级+1段国家级竞赛，不满足')
        currentStar = 4
        overrides.push('5星→4星（项目类型不满足5星条件）')
      }
    }
  }

  // 弱动词不构成深度（仅技能维度，3星以上需 Q17 命中）
  if (dimName === 'skill' && currentStar >= 3) {
    const meetsQ17 = questions['Q17']?.answer === true
    if (!meetsQ17) {
      applied.push('弱动词规则触发：3星以上需强动词（Q17命中），不满足')
      currentStar = 2
      overrides.push(`${'≥3星'}→2星（缺少强动词）`)
    }
  }

  // 评分集中 2-4 星（软约束，不做强制调整，仅记录）
  applied.push('评分集中2-4星（软约束）')

  return { applied, overrides }
}

/**
 * 从 override 列表中解析最终星级
 */
function resolveOverriddenRating(overrides: string[], majorityStar: number): number {
  // overrides 的最后一条即为最终结果
  const last = overrides[overrides.length - 1]
  // 解析最后一个 override 中的最终星级（如 "4星→3星" → 3）
  const match = last.match(/[→>-]\s*(\d)\s*星/)
  if (match) {
    return parseInt(match[1], 10)
  }
  // 无法解析时返回原始多数决结果
  return majorityStar
}
```

```typescript
/**
 * 解析 AI 返回的 31 题答案
 * 缺失的题视为"否"
 */
function parseQuestions(
  raw: unknown,
  expectedKeys: string[]
): Record<string, { answer: boolean; evidence: string | null }> {
  const result: Record<string, { answer: boolean; evidence: string | null }> = {}

  if (!raw || typeof raw !== 'object') {
    // 全部缺失 → 全部"否"
    for (const key of expectedKeys) {
      result[key] = { answer: false, evidence: null }
    }
    return result
  }

  const obj = raw as Record<string, unknown>
  for (const key of expectedKeys) {
    const val = obj[key]
    if (val && typeof val === 'object' && 'answer' in (val as object)) {
      const item = val as { answer: unknown; evidence: unknown }
      result[key] = {
        answer: item.answer === true || item.answer === 'true',
        evidence: typeof item.evidence === 'string' ? item.evidence : null,
      }
    } else {
      // 缺失的题视为"否"
      result[key] = { answer: false, evidence: null }
    }
  }

  return result
}
```

```typescript
  personalProject: number

  const defaults = {

  const n = Number(val)
  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0

function generateScoredComment(
  dimName: 'knowledge' | 'skill',
  scored: ScoredDimension
): string {
  const label = dimName === 'knowledge' ? '知识' : '技能'
  if (scored.rating === 0) return `${label}解析失败`
  const dirSummaries = Object.entries(scored.directions)
    .map(([key, d]) => `${DIRECTION_NAMES[key] || key}${d.star}星`)
    .join('，')
  // 从 adjustments 和 overrides 中提取关键信息
  const notes: string[] = []
  if (scored.voting.adjustments.length > 0) {
    notes.push(scored.voting.adjustments[0])
  }
  if (scored.insurance.overrides.length > 0) {
    notes.push(scored.insurance.overrides[0])
  }
  const noteStr = notes.length > 0 ? `(${notes.join('；')})` : ''
  return `${label}${scored.rating}星：${dirSummaries}${noteStr}`
```
